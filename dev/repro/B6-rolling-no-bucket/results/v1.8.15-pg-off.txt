## rest_filter_inside_bucket
$ POST /cubejs-api/v1/load {"measures": ["orders.revenue_trailing_30d"], "timeDimensions": [{"dimension": "orders.created_at", "granularity": "month", "dateRange": ["2025-04-15", "2025-05-31"]}]}
HTTP 200 [{"orders.created_at.month": "2025-04-01T00:00:00.000", "orders.revenue_trailing_30d": "100", "orders.created_at": "2025-04-01T00:00:00.000"}, {"orders.created_at.month": "2025-05-01T00:00:00.000", "orders.revenue_trailing_30d": "1000", "orders.created_at": "2025-05-01T00:00:00.000"}]
SQL: SELECT q_0."orders__created_at_month", "orders__revenue_trailing_30d" "orders__revenue_trailing_30d" FROM (SELECT "orders.created_at_series"."date_from" "orders__created_at_month", sum("orders__revenue_trailing_30d") "orders__revenue_trailing_30d" FROM (SELECT date_from::timestamp as "date_from", date_to::timestamp as "date_to" FROM (VALUES ('2025-04-01T00:00:00.000', '2025-04-30T23:59:59.999'),('2025-05-01T00:00:00.000', '2025-05-31T23:59:59.999')) AS dates (date_from, date_to)) AS "orders.created_at_series" LEFT JOIN (SELECT date_trunc('day', ("orders_revenue_trailing_30d_cumulative__orders".created_at::timestamptz AT TIME ZONE 'UTC')) "orders__created_at_month", sum("orders_revenue_trailing_30d_cumulative__orders".amount) "orders__revenue_trailing_30d" FROM repro.orders AS "orders_revenue_trailing_30d_cumulative__orders"  WHERE (("orders_revenue_trailing_30d_cumulative__orders".created_at::timestamptz AT TIME ZONE 'UTC') > $1::timestamp - interval '30 days' AND ("orders_revenue_trailing_30d_cumulative__orders".created_at::timestamptz AT TIME ZONE 'UTC') <= $2::timestamp) GROUP BY 1) AS "orders_revenue_trailing_30d_cumulative__base" ON "orders_revenue_trailing_30d_cumulative__base"."orders__created_at_month" > "orders.created_at_series"."date_to" - interval '30 days' AND "orders_revenue_trailing_30d_cumulative__base"."orders__created_at_month" <= "orders.created_at_series"."date_to" GROUP BY 1) AS q_0  ORDER BY 1 ASC LIMIT 10000
