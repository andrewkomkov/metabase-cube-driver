logical_plan | CubeScanWrappedSql
physical_plan | CubeScanExecutionPlan, SQL:
SELECT `q`.`revenue` `revenue`, `q`.`count_uint8_1__` `c` 
FROM (
  SELECT `q`.`revenue` `revenue`, COUNT(*) `count_uint8_1__` 
  FROM (
    SELECT
          `orders`.id `id`, `orders`.amount `amount_dim`, sum(`orders`.amount) `revenue`
        FROM
          repro.orders AS `orders`  GROUP BY `id`, `amount_dim`
  ) AS `q`
  GROUP BY 1
  ORDER BY `q`.`revenue` ASC NULLS LAST
) AS `q`
LIMIT 50000
