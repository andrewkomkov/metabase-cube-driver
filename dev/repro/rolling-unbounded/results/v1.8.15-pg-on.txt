## rest_two_months
$ POST /cubejs-api/v1/load {"measures": ["orders.cumulative_count"], "timeDimensions": [{"dimension": "orders.created_at", "granularity": "month", "dateRange": ["2025-04-01", "2025-05-31"]}]}
HTTP 200 [{"orders.created_at.month": "2025-04-01T00:00:00.000", "orders.cumulative_count": "2", "orders.created_at": "2025-04-01T00:00:00.000"}, {"orders.created_at.month": "2025-05-01T00:00:00.000", "orders.cumulative_count": "3", "orders.created_at": "2025-05-01T00:00:00.000"}]
SQL:  WITH 
time_series AS (  SELECT "d"::timestamptz AS "date_from",
  ("d"::timestamptz + interval '1 month') - interval '1 millisecond' AS "date_to" 
  FROM generate_series('2025-04-01T00:00:00.000'::timestamp, '2025-05-31T23:59:59.999':: timestamp, '1 month'::interval) d ),
cte_0 AS (  SELECT date_trunc('month', ("orders".created_at::timestamptz AT TIME ZONE 'UTC')) "orders__created_at_month", count("orders".id) "orders__cumulative_count" 
  FROM  repro.orders  AS "orders"
  WHERE (("orders".created_at::timestamptz AT TIME ZONE 'UTC') <= $1::timestamptz)
  GROUP BY 1
  ORDER BY  1  ASC),
cte_1 AS (  SELECT "time_series"."date_from" "orders__created_at_month", sum("rolling_source"."orders__cumulative_count") "orders__cumulative_count" 
  FROM  time_series  AS "time_series"
  LEFT JOIN  cte_0  AS "rolling_source" ON "rolling_source"."orders__created_at_month" <= "time_series"."date_to"
  GROUP BY 1
  ORDER BY  1  ASC)
SELECT "fk_aggregate"."orders__created_at_month" "orders__created_at_month", "fk_aggregate"."orders__cumulative_count" "orders__cumulative_count" 
FROM  cte_1  AS "fk_aggregate"
ORDER BY  1  ASC
LIMIT 10000
PARAMS: ["2025-06-30T23:59:58.999"]
