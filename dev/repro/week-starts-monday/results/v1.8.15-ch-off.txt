SQL: SELECT
      toDateTime64(toMonday(toTimeZone(toDateTime64(`orders`.created_at, 0), 'UTC'), 'UTC'), 0, 'UTC') `orders__created_at_week`, sum(`orders`.amount) `orders__revenue`
    FROM
      repro.orders AS `orders`  GROUP BY `orders__created_at_week` ORDER BY `orders__created_at_week` ASC LIMIT 10000
