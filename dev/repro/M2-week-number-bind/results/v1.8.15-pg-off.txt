SELECT "q"."w" "w", "q"."c" "c" 
FROM (
  SELECT "orders"."cast_floor_dated" "w", "orders"."measure_orders_c" "c" 
  FROM (
    SELECT
          CAST(FLOOR((CASE WHEN LOWER('day') IN ('year', 'quarter', 'month') THEN (EXTRACT(YEAR FROM AGE(DATE_TRUNC('day', ("orders".created_at::timestamptz AT TIME ZONE 'UTC')), DATE_TRUNC('day', timestamptz '2001-01-07T00:00:00.000Z'))) * 12 + EXTRACT(MONTH FROM AGE(DATE_TRUNC('day', ("orders".created_at::timestamptz AT TIME ZONE 'UTC')), DATE_TRUNC('day', timestamptz '2001-01-07T00:00:00.000Z')))) / CASE LOWER('day') WHEN 'year' THEN 12 WHEN 'quarter' THEN 3 WHEN 'month' THEN 1 END ELSE EXTRACT(EPOCH FROM DATE_TRUNC('day', ("orders".created_at::timestamptz AT TIME ZONE 'UTC')) - DATE_TRUNC('day', timestamptz '2001-01-07T00:00:00.000Z')) / EXTRACT(EPOCH FROM '1 day'::interval) END::bigint / 7)) AS BIGINT) "cast_floor_dated", count("orders".id) "measure_orders_c"
        FROM
          repro.orders AS "orders"  WHERE ("orders".created_at >= $1::timestamptz) GROUP BY 1
  ) AS "orders"
  ORDER BY "orders"."cast_floor_dated" ASC NULLS LAST
) AS "q"
LIMIT 50000
