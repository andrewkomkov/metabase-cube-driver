## rest_offset_no_rollup
$ POST /cubejs-api/v1/load {"measures": ["orders.revenue_no_rollup"], "timeDimensions": [{"dimension": "orders.created_at", "granularity": "fiscal_quarter"}], "order": {"orders.created_at": "asc"}}
HTTP 200 [{"orders.created_at.fiscal_quarter": "2024-11-01T00:00:00.000", "orders.revenue_no_rollup": "11", "orders.created_at": "2024-11-01T00:00:00.000"}, {"orders.created_at.fiscal_quarter": "2025-02-01T00:00:00.000", "orders.revenue_no_rollup": "1100", "orders.created_at": "2025-02-01T00:00:00.000"}, {"orders.created_at.fiscal_quarter": "2025-05-01T00:00:00.000", "orders.revenue_no_rollup": "110000", "orders.created_at": "2025-05-01T00:00:00.000"}]
SQL: SELECT date_add(month,
        FLOOR(
          date_diff(month, toDateTime64('2026-02-01T00:00:00.000', 3, 'UTC'), toTimeZone(toDateTime64(`orders`.created_at, 0), 'UTC')) /
          date_diff(month, fromUnixTimestamp(0), fromUnixTimestamp(0) + INTERVAL 3 MONTH)
        ) * date_diff(month, fromUnixTimestamp(0), fromUnixTimestamp(0) + INTERVAL 3 MONTH),
        toDateTime64('2026-02-01T00:00:00.000', 3, 'UTC')
    ) `orders__created_at_fiscal_quarter`, sum(amount * 1) `orders__revenue_no_rollup` 
FROM  repro.orders  AS `orders`
GROUP BY 1
ORDER BY  1  ASC
LIMIT 10000
PARAMS: []
